```python
"""Item 12 acceptance: FÖ lessons run late on the einzügig demo seed.

Drives the production HTTP route flow on the einzügig Grundschule
seed once at the production 200 ms LAHC budget, projects the FÖ
placements, and asserts the median position lands in the latter half
of the day. The seed has 6 periods (positions 0..5); latter half is
positions 3..5; median position must be >= 3.
"""

from collections.abc import Awaitable, Callable
from statistics import median

from httpx import AsyncClient
from sqlalchemy import select
from sqlalchemy.ext.asyncio import AsyncSession

from klassenzeit_backend.db.models.lesson import Lesson
from klassenzeit_backend.db.models.scheduled_lesson import ScheduledLesson
from klassenzeit_backend.db.models.school_class import SchoolClass
from klassenzeit_backend.db.models.subject import Subject
from klassenzeit_backend.db.models.user import User
from klassenzeit_backend.db.models.week_scheme import TimeBlock
from klassenzeit_backend.main import app
from klassenzeit_backend.seed.demo_grundschule import seed_demo_grundschule

CreateUserFn = Callable[..., Awaitable[tuple[User, str]]]
LoginFn = Callable[[str, str], Awaitable[None]]


async def test_demo_grundschule_foe_runs_late(
    db_session: AsyncSession,
    client: AsyncClient,
    create_test_user: CreateUserFn,
    login_as: LoginFn,
    monkeypatch: "pytest.MonkeyPatch",
) -> None:
    monkeypatch.setattr(app.state.settings, "solve_deadline_ms", 200)
    await seed_demo_grundschule(db_session)
    await db_session.flush()

    admin, password = await create_test_user(
        email="admin-foe-late@example.com",
        password="foe-late-test-password-12345",  # noqa: S106
        role="admin",
    )
    await login_as(admin.email, password)

    classes = (
        (await db_session.execute(select(SchoolClass).order_by(SchoolClass.grade_level)))
        .scalars()
        .all()
    )
    assert [c.name for c in classes] == ["1a", "2a", "3a", "4a"]

    for school_class in classes:
        gen_resp = await client.post(f"/api/classes/{school_class.id}/generate-lessons")
        assert gen_resp.status_code == 201, gen_resp.text

    for school_class in classes:
        sched_resp = await client.post(f"/api/classes/{school_class.id}/schedule")
        assert sched_resp.status_code == 200, sched_resp.text
        body = sched_resp.json()
        assert body["violations"] == [], (school_class.name, body["violations"])

    foe_subject_id = (
        await db_session.execute(select(Subject.id).where(Subject.short_name == "FÖ"))
    ).scalar_one()

    foe_positions = (
        (
            await db_session.execute(
                select(TimeBlock.position)
                .join(ScheduledLesson, ScheduledLesson.time_block_id == TimeBlock.id)
                .join(Lesson, Lesson.id == ScheduledLesson.lesson_id)
                .where(Lesson.subject_id == foe_subject_id)
            )
        )
        .scalars()
        .all()
    )
    assert foe_positions, "expected at least one FÖ placement after solving"

    median_position = median(foe_positions)
    assert median_position >= 3, (
        f"FÖ placements should run late (median >= 3 on 6-period grid); "
        f"got median={median_position} from positions={sorted(foe_positions)}"
    )
```

`pytest.MonkeyPatch` is referenced as a string for ty so the import at top stays minimal; if `ty` flags it, hoist `import pytest` to the top per `backend/CLAUDE.md`'s "no in-function imports" rule.
